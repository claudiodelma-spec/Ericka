datasource db {
  provider = "sqlite"
  url      = "file:./sandbox.db"
}

generator client {
  provider = "prisma-client-js"
}

model User {
  id        String   @id @default(uuid())
  email     String   @unique
  password  String
  name      String
  roleId    String
  role      Role     @relation(fields: [roleId], references: [id])
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

model Role {
  id          String   @id @default(uuid())
  name        String   @unique
  description String?
  users       User[]
}

model Product {
  id          String   @id @default(uuid())
  name        String
  description String?
  price       Float
  imageUrl    String?
  active      Boolean  @default(true)
  categoryId  String
}

model Order {
  id          String   @id @default(uuid())
  orderNumber String   @unique
  studentName String
  grade       String
  group       String?
  total       Float
  status      String   @default("RECIBIDO")
  createdAt   DateTime @default(now())
}
